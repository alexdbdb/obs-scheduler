cmake_minimum_required(VERSION 3.24)
project(broadcast-scheduler VERSION 0.2.1 LANGUAGES C CXX)
set(BROADCAST_SCHEDULER_VERSION "${PROJECT_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_PLUGIN "Build native OBS module" ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network HttpServer Concurrent)
find_package(SQLite3 REQUIRED)
find_package(LibIcal CONFIG QUIET)
if(TARGET ical)
  set(ICAL_TARGET ical)
elseif(TARGET LibIcal::ical)
  set(ICAL_TARGET LibIcal::ical)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ICAL REQUIRED IMPORTED_TARGET libical)
  set(ICAL_TARGET PkgConfig::ICAL)
endif()
add_library(scheduler-core STATIC
  src/model.cpp src/store.cpp src/scheduler.cpp src/calendar.cpp
  src/providers.cpp src/secrets.cpp src/api.cpp src/runtime.cpp src/messages.cpp)
target_compile_definitions(scheduler-core PUBLIC PLUGIN_VERSION="${BROADCAST_SCHEDULER_VERSION}")
target_include_directories(scheduler-core PUBLIC src)
target_link_libraries(scheduler-core PUBLIC Qt6::Core Qt6::Network Qt6::HttpServer Qt6::Concurrent SQLite::SQLite3 ${ICAL_TARGET})
if(WIN32)
  # The Windows packaging script builds libical with STATIC_ONLY=ON. The
  # upstream config target does not propagate the static-header define, so
  # without this the headers annotate every symbol as __declspec(dllimport)
  # and MSVC/MinGW link fails with unresolved __imp_* symbols.
  target_compile_definitions(scheduler-core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN
    LIBICAL_ICAL_STATIC_DEFINE)
  target_link_libraries(scheduler-core PRIVATE Crypt32)
elseif(APPLE)
  target_link_libraries(scheduler-core PRIVATE "-framework Security" "-framework CoreFoundation")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SECRET IMPORTED_TARGET libsecret-1)
  if(SECRET_FOUND)
    target_link_libraries(scheduler-core PRIVATE PkgConfig::SECRET)
    target_compile_definitions(scheduler-core PRIVATE HAVE_LIBSECRET)
  endif()
endif()
if(BUILD_PLUGIN)
  find_package(libobs CONFIG QUIET)
  if(NOT TARGET OBS::libobs)
    find_path(OBS_INCLUDE obs-module.h HINTS ${OBS_SOURCE_DIR}/libobs PATH_SUFFIXES obs REQUIRED)
    find_library(OBS_LIBRARY NAMES obs libobs.so.0 REQUIRED)
    add_library(OBS::libobs UNKNOWN IMPORTED)
    set_target_properties(OBS::libobs PROPERTIES IMPORTED_LOCATION "${OBS_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${OBS_INCLUDE}")
  endif()
  find_package(obs-frontend-api CONFIG QUIET)
  if(NOT TARGET OBS::obs-frontend-api)
    find_path(OBS_FRONTEND_INCLUDE obs-frontend-api.h HINTS ${OBS_SOURCE_DIR}/frontend/api PATH_SUFFIXES obs)
    find_library(OBS_FRONTEND_LIBRARY NAMES obs-frontend-api libobs-frontend-api.so.0 REQUIRED)
    add_library(OBS::obs-frontend-api UNKNOWN IMPORTED)
    set_target_properties(OBS::obs-frontend-api PROPERTIES IMPORTED_LOCATION "${OBS_FRONTEND_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${OBS_FRONTEND_INCLUDE}")
  endif()
  add_library(broadcast-scheduler MODULE src/plugin.cpp src/obs-adapter.cpp src/ui.cpp)
  target_link_libraries(broadcast-scheduler PRIVATE scheduler-core Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
  target_compile_definitions(broadcast-scheduler PRIVATE PLUGIN_VERSION="${BROADCAST_SCHEDULER_VERSION}")
  set_target_properties(broadcast-scheduler PROPERTIES PREFIX "")
  if(WIN32)
    install(TARGETS broadcast-scheduler RUNTIME DESTINATION broadcast-scheduler/bin/64bit LIBRARY DESTINATION broadcast-scheduler/bin/64bit)
    install(DIRECTORY data/ DESTINATION broadcast-scheduler/data)
  elseif(APPLE)
    set_target_properties(broadcast-scheduler PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin)
    install(TARGETS broadcast-scheduler LIBRARY DESTINATION broadcast-scheduler.plugin/Contents/MacOS)
    install(DIRECTORY data/ DESTINATION broadcast-scheduler.plugin/Contents/Resources)
  else()
    install(TARGETS broadcast-scheduler LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
    install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/broadcast-scheduler)
  endif()
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(scheduler-tests tests/test_scheduler.cpp)
  target_link_libraries(scheduler-tests PRIVATE scheduler-core Qt6::Test)
  add_test(NAME scheduler-tests COMMAND scheduler-tests)
  add_executable(scheduler-service tests/service.cpp)
  target_link_libraries(scheduler-service PRIVATE scheduler-core)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME locale-check COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_locales.py)
    add_test(NAME api-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_integration.py $<TARGET_FILE:scheduler-service>)
    set_tests_properties(api-integration PROPERTIES TIMEOUT 30)
  endif()
  if(WIN32)
    add_executable(package-probe tests/package_probe.cpp)
    target_link_libraries(package-probe PRIVATE Qt6::Core Qt6::Network)
  endif()
endif()
