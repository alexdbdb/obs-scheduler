cmake_minimum_required(VERSION 3.24)
project(scheduler-sqlite LANGUAGES C)
file(GLOB sqlite_sources "${SQLITE_SOURCE_DIR}/sqlite3.c")
if(NOT sqlite_sources)
  message(FATAL_ERROR "SQLITE_SOURCE_DIR must contain sqlite3.c and sqlite3.h")
endif()
add_library(sqlite3 STATIC ${sqlite_sources})
target_include_directories(sqlite3 PUBLIC $<BUILD_INTERFACE:${SQLITE_SOURCE_DIR}> $<INSTALL_INTERFACE:include>)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
install(TARGETS sqlite3 ARCHIVE DESTINATION lib)
install(FILES ${SQLITE_SOURCE_DIR}/sqlite3.h DESTINATION include)
